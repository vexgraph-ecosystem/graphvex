cmake_minimum_required(VERSION 3.24)
# Standalone CLion metadata only. Actual builds belong to b; no fetch or link.
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Supported OS floor")
endif()
project(graphvex_ide LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS YES)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VEXSPOKE_SOURCE_DIR "" CACHE PATH "Local Vexspoke src; never downloaded")
set(VEXGRAPH_SHADER_DIR "" CACHE PATH "Existing b shader output containing quad_spv.h; never generated here")
find_path(VULKAN_INCLUDE_DIR vulkan/vulkan.h DOC "Local Vulkan SDK headers; never downloaded")
file(GLOB_RECURSE sources CONFIGURE_DEPENDS "src/*.c" "src/*.h")
list(FILTER sources EXCLUDE REGEX "/(attic|vendor|third_party|_old)/")
add_library(graphvex_ide OBJECT EXCLUDE_FROM_ALL ${sources})
target_include_directories(graphvex_ide PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
foreach(path VEXSPOKE_SOURCE_DIR VEXGRAPH_SHADER_DIR VULKAN_INCLUDE_DIR)
    if(${path})
        target_include_directories(graphvex_ide PRIVATE "${${path}}")
    endif()
endforeach()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(graphvex_ide PRIVATE -std=gnu23 -Wall -Wextra -Werror)
    if(APPLE)
        target_compile_options(graphvex_ide PRIVATE -mcpu=apple-m1)
    endif()
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    execute_process(COMMAND "${CMAKE_C_COMPILER}" -print-resource-dir
        OUTPUT_VARIABLE resource OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(EXISTS "${resource}/include/stddef.h")
        target_compile_options(graphvex_ide PRIVATE -isystem "${resource}/include")
    endif()
endif()
message(STATUS "IDE metadata only: excluded object target. Use b for builds and shaders; missing headers stay errors.")
